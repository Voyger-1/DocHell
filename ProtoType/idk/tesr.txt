// Regex to remove <mc:Choice>...</mc:Choice>
var mcChoiceRe = regexp.MustCompile(`(?s)<mc:Choice[^>]*>.*?</mc:Choice>`)

// RemoveMCChoices cleans XML content by stripping <mc:Choice> blocks
func RemoveMCChoices(xmlData []byte) []byte {
	fmt.Println("Cleaning mc:Choice blocks...", xmlData)
	return mcChoiceRe.ReplaceAll(xmlData, []byte(""))
}

// CleanPPTX loads an input PPTX, removes mc:Choice blocks from all XML parts,
// and writes a new PPTX to outPath.
func CleanPPTX(inPath, outPath string) error {
	// Open input pptx (zip archive)
	r, err := zip.OpenReader(inPath)
	if err != nil {
		return fmt.Errorf("open pptx: %w", err)
	}
	defer r.Close()

	// Create output pptx
	outFile, err := os.Create(outPath)
	if err != nil {
		return fmt.Errorf("create out pptx: %w", err)
	}
	defer outFile.Close()

	w := zip.NewWriter(outFile)
	defer w.Close()

	// Process each file inside pptx
	for _, f := range r.File {
		rc, err := f.Open()
		if err != nil {
			return fmt.Errorf("open inner file: %w", err)
		}

		var buf bytes.Buffer
		if _, err := io.Copy(&buf, rc); err != nil {
			rc.Close()
			return fmt.Errorf("copy inner file: %w", err)
		}
		rc.Close()

		data := buf.Bytes()

		// Only clean XML files
		if filepath.Ext(f.Name) == ".xml" {
			data = RemoveMCChoices(data)
		}

		// Write file back into new zip
		hdr := f.FileHeader
		hdr.Method = zip.Deflate
		wr, err := w.CreateHeader(&hdr)
		if err != nil {
			return fmt.Errorf("create zip entry: %w", err)
		}
		if _, err := wr.Write(data); err != nil {
			return fmt.Errorf("write zip entry: %w", err)
		}
	}

	return nil
}

func main() {
	inPath := "example.pptx"
	outPath := "clesssaned.pptx"

	if err := CleanPPTX(inPath, outPath); err != nil {
		panic(err)
	}

	fmt.Println("✅ Cleaned PPTX written to:", outPath)
}

// --- (types unchanged) ---
type Link struct {
	Href   string `json:"href"`
	Text   string `json:"text"`
	X      int    `json:"x"`
	Y      int    `json:"y"`
	Width  int    `json:"width"`
	Height int    `json:"height"`
}

type TextElement struct {
	Text   string `json:"text"`
	X      int    `json:"x"`
	Y      int    `json:"y"`
	Width  int    `json:"width"`
	Height int    `json:"height"`
}

type Media struct {
	File     string `json:"file"`
	Type     string `json:"type"`
	X        int    `json:"x"`
	Y        int    `json:"y"`
	Width    int    `json:"width"`
	Height   int    `json:"height"`
	Rotation int    `json:"rotation"`
	FlipH    bool   `json:"flipH"`
	FlipV    bool   `json:"flipV"`
}

type Slide struct {
	Slide      string        `json:"slide"`
	Links      []Link        `json:"links"`
	Media      []Media       `json:"media"`

}

	Author      string `json:"author"`

	Company     string `json:"company"`
}

type Result struct {
	Metadata Metadata `json:"metadata"`

}

	Id     string

// --- (constants unchanged) ---
const (
	defaultSlideWidthEMU  = 96
	defaultSlideHeightEMU = 96
	TargetWidthPX         = 1280
	TargetHeightPX        = 720
)

var (
	slideWidthEMU  = defaultSlideWidthEMU
	slideHeightEMU = defaultSlideHeightEMU

	offRe   = regexp.MustCompile(`<a:off[^>]*x="(\d+)"[^>]*y="(\d+)"`)
	extRe   = regexp.MustCompile(`<a:ext[^>]*cx="(\d+)"[^>]*cy="(\d+)"`)
	rotRe   = regexp.MustCompile(`rot="(-?\d+)"`)
	flipHRe = regexp.MustCompile(`flipH="1"`)
	flipVRe = regexp.MustCompile(`flipV="1"`)
	rIdRe   = regexp.MustCompile(`r:(?:embed|link)\s*=\s*"(rId\d+)"`)
)

// readZipFile unchanged
func readZipFile(f *zip.File) string {
	rc, _ := f.Open()
	if rc == nil {
		return ""
	}
	defer rc.Close()
	buf, _ := io.ReadAll(rc)
	return string(buf)
}

func emuToPxX(emu string) int {
	val, _ := strconv.ParseInt(emu, 10, 64)
	return int(float64(val) / float64(slideWidthEMU) * float64(TargetWidthPX))
}

func emuToPxY(emu string) int {
	val, _ := strconv.ParseInt(emu, 10, 64)
	return int(float64(val) / float64(slideHeightEMU) * float64(TargetHeightPX))
}

func getCoordsNear(xmlStr string, matchPos int) (x, y, w, h, rotation int, flipH, flipV bool) {
	parentStart := strings.LastIndex(xmlStr[:matchPos], "<p:")
	if parentStart < 0 {
		parentStart = matchPos - 2000
		if parentStart < 0 {
			parentStart = 0
		}
	}

	windowBefore := xmlStr[parentStart:matchPos]
	windowAfterEnd := matchPos + 2000
	if windowAfterEnd > len(xmlStr) {
		windowAfterEnd = len(xmlStr)
	}
	windowAfter := xmlStr[matchPos:windowAfterEnd]

	// Position and size
	if offMatches := offRe.FindAllStringSubmatch(windowBefore, -1); len(offMatches) > 0 {
		last := offMatches[len(offMatches)-1]
		x = emuToPxX(last[1])
		y = emuToPxY(last[2])
	}
	if extMatches := extRe.FindAllStringSubmatch(windowBefore, -1); len(extMatches) > 0 {
		last := extMatches[len(extMatches)-1]
		w = emuToPxX(last[1])
		h = emuToPxY(last[2])
	}

	// Rotation
	if rotMatches := rotRe.FindStringSubmatch(windowBefore); len(rotMatches) > 1 {
		raw, _ := strconv.Atoi(rotMatches[1])
		rotation = raw / 60000
	}

	// Flips
	flipH = flipHRe.MatchString(windowBefore)
	flipV = flipVRe.MatchString(windowBefore)

	// Fallback to windowAfter if missing
	if x == 0 && y == 0 {
		if offMatches := offRe.FindAllStringSubmatch(windowAfter, -1); len(offMatches) > 0 {
			first := offMatches[0]
			x = emuToPxX(first[1])
			y = emuToPxY(first[2])
		}
	}
	if w == 0 && h == 0 {
		if extMatches := extRe.FindAllStringSubmatch(windowAfter, -1); len(extMatches) > 0 {
			first := extMatches[0]
			w = emuToPxX(first[1])
			h = emuToPxY(first[2])
		}
	}
	if rotation == 0 {
		if rotMatches := rotRe.FindStringSubmatch(windowAfter); len(rotMatches) > 1 {
			raw, _ := strconv.Atoi(rotMatches[1])
			rotation = raw / 60000
		}
	}
	if !flipH {
		flipH = flipHRe.MatchString(windowAfter)
	}
	if !flipV {
		flipV = flipVRe.MatchString(windowAfter)
	}

	return
}

func exportFile(target string, files map[string]*zip.File, outDir string) string {
	target = strings.TrimPrefix(target, "../")
	key := "ppt/" + target
	if f, ok := files[key]; ok {
		os.MkdirAll(outDir, 0755)
		outPath := filepath.Join(outDir, filepath.Base(target))
		rc, _ := f.Open()
		if rc == nil {
			return ""
		}
		defer rc.Close()
		out, _ := os.Create(outPath)
		defer out.Close()
		io.Copy(out, rc)
		return outPath
	}
	if f, ok := files[target]; ok {
		os.MkdirAll(outDir, 0755)
		outPath := filepath.Join(outDir, filepath.Base(target))
		rc, _ := f.Open()
		if rc == nil {
			return ""
		}
		defer rc.Close()
		out, _ := os.Create(outPath)
		defer out.Close()
		io.Copy(out, rc)
		return outPath
	}
	return ""
}

func extractPPTX(pptxPath string, outJSON string) error {
	r, err := zip.OpenReader(pptxPath)
	if err != nil {
		return fmt.Errorf("cannot open pptx: %v", err)
	}
	defer r.Close()

	files := map[string]*zip.File{}
	for _, f := range r.File {
		files[f.Name] = f
	}

	var result Result
	result.Metadata = Metadata{}

	// Slide dimensions
	if presF, ok := files["ppt/presentation.xml"]; ok {
		presXML := readZipFile(presF)
		if m := regexp.MustCompile(`<p:sldSz[^>]*cx="(\d+)"[^>]*cy="(\d+)"`).FindStringSubmatch(presXML); m != nil {
			if v, err := strconv.Atoi(m[1]); err == nil && v > 0 {
				slideWidthEMU = v
			}
			if v, err := strconv.Atoi(m[2]); err == nil && v > 0 {
				slideHeightEMU = v
			}
		}
	}

	// Metadata
	if f, ok := files["docProps/core.xml"]; ok {
		data := readZipFile(f)
		if m := regexp.MustCompile(`<dc:creator>(.*?)</dc:creator>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Author = m[1]
		}
		if m := regexp.MustCompile(`<dc:title>(.*?)</dc:title>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Title = m[1]
		}
	}
	if f, ok := files["docProps/app.xml"]; ok {
		data := readZipFile(f)
		if m := regexp.MustCompile(`<Slides>(\d+)</Slides>`).FindStringSubmatch(data); m != nil {
			result.Metadata.SlideCount = m[1]
		}
		if m := regexp.MustCompile(`<Application>(.*?)</Application>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Application = m[1]
		}
		if m := regexp.MustCompile(`<Company>(.*?)</Company>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Company = m[1]
		}
	}

	os.MkdirAll("output_media", 0755)

	// Slide loop
	for i := 1; ; i++ {
		slidePath := fmt.Sprintf("ppt/slides/slide%d.xml", i)
		f, ok := files[slidePath]
		if !ok {
			break
		}
		xmlStr := readZipFile(f)

		relsPath := fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels", i)
		rels := map[string]Relationship{}
		if rf, ok := files[relsPath]; ok {
			relsXML := readZipFile(rf)
			relRe := regexp.MustCompile(`<Relationship[^>]*Id="([^"]+)"[^>]*Type="([^"]+)"[^>]*Target="([^"]+)"[^>]*\/?>`)
			for _, m := range relRe.FindAllStringSubmatch(relsXML, -1) {
				rels[m[1]] = Relationship{Id: m[1], Type: m[2], Target: m[3]}
			}
		}

		var links []Link
		var mediaList []Media
		var textBlocks []TextElement

		// Extract links and text blocks
		spRe := regexp.MustCompile(`(?s)<p:sp.*?>.*?</p:sp>`)
		for _, sp := range spRe.FindAllString(xmlStr, -1) {
			text := ""
			if m := regexp.MustCompile(`<a:t>(.*?)</a:t>`).FindAllStringSubmatch(sp, -1); m != nil {
				for _, mm := range m {
					text += mm[1]
				}
			}
			href := ""
			if m := regexp.MustCompile(`a:hlinkClick[^>]*r:id="(rId\d+)"`).FindStringSubmatch(sp); m != nil {
				if rel, ok := rels[m[1]]; ok {
					href = rel.Target
				}
			}
			x, y, w, h, _, _, _ := getCoordsNear(xmlStr, strings.Index(xmlStr, sp)+len(sp)/2)
			if href != "" || text != "" {
				links = append(links, Link{Href: href, Text: text, X: x, Y: y, Width: w, Height: h})
			}
			// Extract all text blocks for potential removal
			if text != "" {
				textBlocks = append(textBlocks, TextElement{Text: text, X: x, Y: y, Width: w, Height: h})
			}
		}

		// Extract media
		for _, match := range rIdRe.FindAllStringSubmatchIndex(xmlStr, -1) {
			if len(match) < 4 {
				continue
			}
			rid := xmlStr[match[2]:match[3]]
			matchPos := match[0]
			x, y, w, h, rotation, flipH, flipV := getCoordsNear(xmlStr, matchPos)

			if rel, ok := rels[rid]; ok {
				ltype := strings.ToLower(rel.Type)
				ext := strings.ToLower(filepath.Ext(rel.Target))
				mediaKind := ""
				if strings.Contains(ltype, "image") || strings.Contains(ext, "png") || strings.Contains(ext, "jpg") {
					mediaKind = "image"
				} else if strings.Contains(ltype, "video") || strings.Contains(ext, "mp4") {
					mediaKind = "video"
				} else if strings.Contains(ltype, "audio") || strings.Contains(ext, "mp3") {
					mediaKind = "audio"
				}

				if mediaKind != "" {
					local := exportFile(rel.Target, files, "output_media")
					if local != "" {
						mediaList = append(mediaList, Media{
							File:     local,
							Type:     mediaKind,
							X:        x,
							Y:        y,
							Width:    w,
							Height:   h,
							Rotation: rotation,
							FlipH:    flipH,
							FlipV:    flipV,
						})
					}
				}
			}
		}

		result.Slides = append(result.Slides, Slide{
			Slide:      fmt.Sprint(i),
			Links:      links,
			Media:      mediaList,
			TextBlocks: textBlocks,
		})
	}

	out, _ := json.MarshalIndent(result, "", "  ")
	return os.WriteFile(outJSON, out, 0644)
}

// --- New helper: sanitize XML content by replacing AlternateContent with fallback ---
// This will repeatedly replace <mc:AlternateContent> blocks that contain <mc:Choice> with the inner <mc:Fallback> content.
// It also removes stray <mc:Choice>...</mc:Choice> tags if present outside of an AlternateContent wrapper.
func sanitizeXML(xmlStr string) string {
	// 1) Replace entire <mc:AlternateContent ...> ... <mc:Fallback>FALLBACK</mc:Fallback> ... </mc:AlternateContent>
	// with the FALLBACK content.
	// Use (?s) to allow dot to match newlines and non-greedy matches.
	acRe := regexp.MustCompile(`(?s)<mc:AlternateContent\b[^>]*>.*?<mc:Fallback\b[^>]*>(.*?)</mc:Fallback>.*?</mc:AlternateContent>`)
	changed := true
	for changed {
		changed = false
		if acRe.MatchString(xmlStr) {
			xmlStr = acRe.ReplaceAllString(xmlStr, "$1")
			changed = true
		}
	}

	// 2) Remove any remaining <mc:Choice ...>...</mc:Choice> blocks (if any)
	choiceRe := regexp.MustCompile(`(?s)<mc:Choice\b[^>]*>.*?</mc:Choice>`)
	xmlStr = choiceRe.ReplaceAllString(xmlStr, "")

	// 3) Remove mc:AlternateContent tags without fallback (just in case)
	acTagsRe := regexp.MustCompile(`(?s)<mc:AlternateContent\b[^>]*>(.*?)</mc:AlternateContent>`)
	xmlStr = acTagsRe.ReplaceAllString(xmlStr, "$1")

	return xmlStr
}

// Clean PPTX by removing text that overlaps with media and convert to PDF
func cleanPPTXAndConvert(pptxPath, metaJSONPath, outPDF string) error {
	// Read metadata
	metaData, err := os.ReadFile(metaJSONPath)
	if err != nil {
		return fmt.Errorf("cannot read meta.json: %v", err)
	}

	var result Result
	if err := json.Unmarshal(metaData, &result); err != nil {
		return fmt.Errorf("cannot parse meta.json: %v", err)
	}

	// Open original PPTX
	r, err := zip.OpenReader(pptxPath)
	if err != nil {
		return fmt.Errorf("cannot open pptx: %v", err)
	}
	defer r.Close()

	// Prepare cleaned output path and remove previous outputs if present
	outputPath := pptxPath[:len(pptxPath)-5] + "_cleaned.pptx"
	// Remove previous cleaned PPTX if exists
	if _, err := os.Stat(outputPath); err == nil {
		if err := os.Remove(outputPath); err != nil {
			return fmt.Errorf("failed to remove previous cleaned pptx: %v", err)
		}
	}
	// Remove previous output PDF if exists
	if _, err := os.Stat(outPDF); err == nil {
		if err := os.Remove(outPDF); err != nil {
			return fmt.Errorf("failed to remove previous output pdf: %v", err)
		}
	}

	outputFile, err := os.Create(outputPath)
	if err != nil {
		return fmt.Errorf("cannot create output file: %v", err)
	}
	defer outputFile.Close()

	writer := zip.NewWriter(outputFile)
	defer writer.Close()

	// Process each file in the PPTX
	for _, file := range r.File {
		content, err := file.Open()
		if err != nil {
			continue
		}
		rawBytes, err := io.ReadAll(content)
		content.Close()
		if err != nil {
			continue
		}
		text := string(rawBytes)

		// If this is an XML file inside ppt/ (slides, layouts, masters, notes, etc.), sanitize mc:Choice blocks.
		if strings.HasSuffix(strings.ToLower(file.Name), ".xml") && strings.HasPrefix(file.Name, "ppt/") {
			text = sanitizeXML(text)
		}

		// If this is a slide file, apply the overlapping text removal logic
		if strings.Contains(file.Name, "ppt/slides/slide") {
			slideNum := extractSlideNumber(file.Name)
			if slideNum > 0 && slideNum <= len(result.Slides) {
				slide := result.Slides[slideNum-1]
				text = cleanSlideContent(text, slide)
			}
		}

		// Write (possibly modified) content into the new zip
		header := &zip.FileHeader{
			Name:   file.Name,
			Method: zip.Deflate,
		}
		header.SetModTime(file.Modified)
		outputFileEntry, err := writer.CreateHeader(header)
		if err != nil {
			continue
		}
		_, _ = outputFileEntry.Write([]byte(text))
	}

	// Close writer and file (defer handles writer.Close but ensure flush)
	writer.Close()
	outputFile.Close()

	// Convert cleaned PPTX to PDF using LibreOffice
	fmt.Println("Converting cleaned PPTX to PDF...")
	// Ensure outdir exists
	outdir := filepath.Dir(outPDF)
	if outdir == "" {
		outdir = "."
	}
	// Remove any stray generated pdf for the cleaned file name (libreoffice will generate a pdf matching cleaned base name)
	generatedPDFPath := filepath.Join(outdir, strings.TrimSuffix(filepath.Base(outputPath), ".pptx")+".pdf")
	if _, err := os.Stat(generatedPDFPath); err == nil {
		os.Remove(generatedPDFPath)
	}

	cmd := exec.Command("libreoffice", "--headless", "--convert-to", "pdf", "--outdir", outdir, outputPath)
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("failed to convert to PDF: %v", err)
	}

	// Move the generated PDF to the desired location (if libreoffice wrote to outdir)
	if _, err := os.Stat(generatedPDFPath); err == nil {
		if err := os.Rename(generatedPDFPath, outPDF); err != nil {
			// If move fails, try a copy fallback
			in, err := os.Open(generatedPDFPath)
			if err == nil {
				defer in.Close()
				out, err := os.Create(outPDF)
				if err == nil {
					defer out.Close()
					io.Copy(out, in)
					os.Remove(generatedPDFPath)
				}
			}
		}
	}

	// Clean up the cleaned PPTX (optional - keep for debugging if desired)
	// os.Remove(outputPath)

	return nil
}

// Clean slide content by removing only text that overlaps with media
func cleanSlideContent(xmlStr string, slide Slide) string {
	result := xmlStr
	removedCount := 0

	fmt.Printf("Processing slide with %d text blocks and %d media elements\n", len(slide.TextBlocks), len(slide.Media))

	// Only remove text that overlaps with media coordinates
	for _, textBlock := range slide.TextBlocks {
		hasOverlap := false

		// Check if this text block overlaps with any media
		for _, media := range slide.Media {
			if rectanglesOverlap(textBlock, media) {
				hasOverlap = true
				fmt.Printf("Found overlap: Text '%s' at (%d,%d) overlaps with media at (%d,%d)\n",
					textBlock.Text, textBlock.X, textBlock.Y, media.X, media.Y)
				break
			}
		}

		// Only remove text if it overlaps with media
		if hasOverlap {
			result = removeTextAtCoordinates(result, textBlock)
			removedCount++
		} else {
			fmt.Printf("Keeping text '%s' at (%d,%d) - no overlap with media\n",
				textBlock.Text, textBlock.X, textBlock.Y)
		}
	}

	fmt.Printf("Removed %d overlapping text elements\n", removedCount)
	return result
}

// Remove text content from XML at specific coordinates
func removeTextAtCoordinates(xmlStr string, textBlock TextElement) string {
	// Convert pixel coordinates to EMU for comparison with PPTX XML
	textXEMU := pixelsToEMU(float64(textBlock.X))
	textYEMU := pixelsToEMU(float64(textBlock.Y))
	textWidthEMU := pixelsToEMU(float64(textBlock.Width))
	textHeightEMU := pixelsToEMU(float64(textBlock.Height))

	fmt.Printf("Looking for text at EMU coordinates: x=%d, y=%d, w=%d, h=%d\n",
		textXEMU, textYEMU, textWidthEMU, textHeightEMU)

	// Pattern to match <a:t>text</a:t> and replace with <a:t></a:t>
	// This will remove the text content while keeping the structure
	pattern := fmt.Sprintf(`<a:t>%s</a:t>`, regexp.QuoteMeta(textBlock.Text))
	re := regexp.MustCompile(pattern)
	result := re.ReplaceAllString(xmlStr, "<a:t></a:t>")

	if result != xmlStr {
		fmt.Printf("Successfully removed text: '%s'\n", textBlock.Text)
	} else {
		fmt.Printf("Text not found in XML: '%s'\n", textBlock.Text)
	}

	return result
}

// Convert pixels to EMU (English Metric Units)
// 1 inch = 96 pixels, 1 inch = 914400 EMU
func pixelsToEMU(pixels float64) int64 {
	return int64(pixels * 914400 / 96)
}

// Convert EMU to pixels
func emuToPixels(emu int64) float64 {
	return float64(emu) * 96 / 914400
}

// Check if two rectangles overlap (converting pixel coordinates to EMU for comparison)
func rectanglesOverlap(text TextElement, media Media) bool {
	// Convert pixel coordinates to EMU for comparison with PPTX XML
	textXEMU := pixelsToEMU(float64(text.X))
	textYEMU := pixelsToEMU(float64(text.Y))
	textWidthEMU := pixelsToEMU(float64(text.Width))
	textHeightEMU := pixelsToEMU(float64(text.Height))

	mediaXEMU := pixelsToEMU(float64(media.X))
	mediaYEMU := pixelsToEMU(float64(media.Y))
	mediaWidthEMU := pixelsToEMU(float64(media.Width))
	mediaHeightEMU := pixelsToEMU(float64(media.Height))

	return !(textXEMU+textWidthEMU <= mediaXEMU ||
		mediaXEMU+mediaWidthEMU <= textXEMU ||
		textYEMU+textHeightEMU <= mediaYEMU ||
		mediaYEMU+mediaHeightEMU <= textYEMU)
}

// Extract slide number from filename
func extractSlideNumber(filename string) int {
	re := regexp.MustCompile(`slide(\d+)\.xml`)
	matches := re.FindStringSubmatch(filename)
	if len(matches) > 1 {
		if num, err := strconv.Atoi(matches[1]); err == nil && num > 0 {
			return num
		}
	}
	return 0
}

// Copy file to zip (still available in case you want to use it elsewhere)
func copyFileToZip(writer *zip.Writer, file *zip.File) error {
	content, err := file.Open()
	if err != nil {
		return err
	}
	defer content.Close()

	header := &zip.FileHeader{
		Name:   file.Name,
		Method: zip.Deflate,
	}
	header.SetModTime(file.Modified)

	outputFile, err := writer.CreateHeader(header)
	if err != nil {
		return err
	}

	_, err = io.Copy(outputFile, content)
	return err
}
