type Link struct {
	Href   string `json:"href"`
	Text   string `json:"text"`
	X      int    `json:"x"`
	Y      int    `json:"y"`
	Width  int    `json:"width"`
	Height int    `json:"height"`
}

type Media struct {
	File   string `json:"file"`
	Type   string `json:"type"`
	X      int    `json:"x"`
	Y      int    `json:"y"`
	Width  int    `json:"width"`
	Height int    `json:"height"`
}

type Slide struct {
	Slide string  `json:"slide"`
	Links []Link  `json:"links"`
	Media []Media `json:"media"`
}

type Metadata struct {
	Author      string `json:"author"`
	Title       string `json:"title"`
	SlideCount  string `json:"slideCount"`
	Application string `json:"application"`
	Company     string `json:"company"`
}

type Result struct {
	Metadata Metadata `json:"metadata"`
	Slides   []Slide  `json:"slides"`
}

type Relationship struct {
	Id     string
	Target string
	Type   string
}

const (
	defaultSlideWidthEMU  = 9144000
	defaultSlideHeightEMU = 6858000
)

const (
	TargetWidthPX  = 600
	TargetHeightPX = 400
)

var (
	slideWidthEMU  = defaultSlideWidthEMU
	slideHeightEMU = defaultSlideHeightEMU

	offRe = regexp.MustCompile(`<a:off[^>]*x="(\d+)"[^>]*y="(\d+)"`)
	extRe = regexp.MustCompile(`<a:ext[^>]*cx="(\d+)"[^>]*cy="(\d+)"`)

	rIdRe = regexp.MustCompile(`r:(?:embed|link)\s*=\s*"(rId\d+)"`)

	// shape regexes used for removal pass
	spRe    = regexp.MustCompile(`(?s)<p:sp.*?</p:sp>`)
	cxnRe   = regexp.MustCompile(`(?s)<p:cxnSp.*?</p:cxnSp>`)
	picRe   = regexp.MustCompile(`(?s)<p:pic.*?</p:pic>`)
	aTRe    = regexp.MustCompile(`<a:t[^>]*>.*?</a:t>`)
	sldSzRe = regexp.MustCompile(`<p:sldSz[^>]*cx="(\d+)"[^>]*cy="(\d+)"`)
)

func readZipFile(f *zip.File) string {
	rc, _ := f.Open()
	if rc == nil {
		return ""
	}
	defer rc.Close()
	buf, _ := io.ReadAll(rc)
	return string(buf)
}

func emuToPxX(emu string) int {
	val, _ := strconv.ParseInt(emu, 10, 64)
	if slideWidthEMU == 0 {
		slideWidthEMU = defaultSlideWidthEMU
	}
	return int(float64(val) / float64(slideWidthEMU) * float64(TargetWidthPX))
}

func emuToPxY(emu string) int {
	val, _ := strconv.ParseInt(emu, 10, 64)
	if slideHeightEMU == 0 {
		slideHeightEMU = defaultSlideHeightEMU
	}
	return int(float64(val) / float64(slideHeightEMU) * float64(TargetHeightPX))
}

func getCoordsNear(xmlStr string, matchPos int) (int, int, int, int) {
	parentStart := strings.LastIndex(xmlStr[:matchPos], "<p:")
	if parentStart < 0 {
		parentStart = matchPos - 2000
		if parentStart < 0 {
			parentStart = 0
		}
	}

	windowBefore := xmlStr[parentStart:matchPos]
	windowAfterEnd := matchPos + 2000
	if windowAfterEnd > len(xmlStr) {
		windowAfterEnd = len(xmlStr)
	}
	windowAfter := xmlStr[matchPos:windowAfterEnd]

	x, y, w, h := 0, 0, 0, 0

	if offMatches := offRe.FindAllStringSubmatch(windowBefore, -1); len(offMatches) > 0 {
		last := offMatches[len(offMatches)-1]
		x = emuToPxX(last[1])
		y = emuToPxY(last[2])
	}
	if extMatches := extRe.FindAllStringSubmatch(windowBefore, -1); len(extMatches) > 0 {
		last := extMatches[len(extMatches)-1]
		w = emuToPxX(last[1])
		h = emuToPxY(last[2])
	}

	if x == 0 && y == 0 {
		if offMatches := offRe.FindAllStringSubmatch(windowAfter, -1); len(offMatches) > 0 {
			first := offMatches[0]
			x = emuToPxX(first[1])
			y = emuToPxY(first[2])
		}
	}
	if w == 0 && h == 0 {
		if extMatches := extRe.FindAllStringSubmatch(windowAfter, -1); len(extMatches) > 0 {
			first := extMatches[0]
			w = emuToPxX(first[1])
			h = emuToPxY(first[2])
		}
	}

	return x, y, w, h
}

func exportFile(target string, files map[string]*zip.File, outDir string) string {
	target = strings.TrimPrefix(target, "../")
	key := "ppt/" + target
	if f, ok := files[key]; ok {
		os.MkdirAll(outDir, 0755)
		outPath := filepath.Join(outDir, filepath.Base(target))
		rc, _ := f.Open()
		if rc == nil {
			return ""
		}
		defer rc.Close()
		out, _ := os.Create(outPath)
		if out == nil {
			return ""
		}
		defer out.Close()
		io.Copy(out, rc)
		return outPath
	}
	if f, ok := files[target]; ok {
		os.MkdirAll(outDir, 0755)
		outPath := filepath.Join(outDir, filepath.Base(target))
		rc, _ := f.Open()
		if rc == nil {
			return ""
		}
		defer rc.Close()
		out, _ := os.Create(outPath)
		if out == nil {
			return ""
		}
		defer out.Close()
		io.Copy(out, rc)
		return outPath
	}
	return ""
}

// unzip into dest
func unzip(src, dest string) error {
	r, err := zip.OpenReader(src)
	if err != nil {
		return err
	}
	defer r.Close()

	for _, f := range r.File {
		targetPath := filepath.Join(dest, f.Name)
		if f.FileInfo().IsDir() {
			if err := os.MkdirAll(targetPath, 0755); err != nil {
				return err
			}
			continue
		}
		if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
			return err
		}
		rc, err := f.Open()
		if err != nil {
			return err
		}
		outFile, err := os.Create(targetPath)
		if err != nil {
			rc.Close()
			return err
		}
		_, err = io.Copy(outFile, rc)
		rc.Close()
		outFile.Close()
		if err != nil {
			return err
		}
	}
	return nil
}

// zipDir to pptx
func zipDir(source, target string) error {
	outFile, err := os.Create(target)
	if err != nil {
		return err
	}
	defer outFile.Close()
	zw := zip.NewWriter(outFile)
	defer zw.Close()

	// walk
	return filepath.Walk(source, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		rel, err := filepath.Rel(source, path)
		if err != nil {
			return err
		}
		rel = filepath.ToSlash(rel)
		if info.IsDir() {
			if rel == "." {
				return nil
			}
			_, err := zw.Create(rel + "/")
			return err
		}
		fw, err := zw.Create(rel)
		if err != nil {
			return err
		}
		fr, err := os.Open(path)
		if err != nil {
			return err
		}
		defer fr.Close()
		_, err = io.Copy(fw, fr)
		return err
	})
}

// overlap check
func overlap(ax, ay, aw, ah, bx, by, bw, bh int) bool {
	return ax < bx+bw && ax+aw > bx && ay < by+bh && ay+ah > by
}

// remove shapes in slide XML that overlap any rect in rects
func removeShapesFromSlideXML(xmlStr string, rects [][4]int) (string, int) {
	removed := 0
	// remove <p:sp>
	shapes := spRe.FindAllString(xmlStr, -1)
	for _, sp := range shapes {
		// only care if it has text (a:t)
		if !aTRe.MatchString(sp) {
			continue
		}
		pos := strings.Index(xmlStr, sp)
		if pos < 0 {
			continue
		}
		x, y, w, h := getCoordsNear(xmlStr, pos+len(sp)/2)
		if x == 0 && y == 0 && w == 0 && h == 0 {
			// couldn't parse coords - skip to avoid accidental removal
			continue
		}
		over := false
		for _, r := range rects {
			if overlap(x, y, w, h, r[0], r[1], r[2], r[3]) {
				over = true
				break
			}
		}
		if over {
			xmlStr = strings.Replace(xmlStr, sp, "", 1)
			removed++
		}
	}

	// remove connectors <p:cxnSp>
	conns := cxnRe.FindAllString(xmlStr, -1)
	for _, cn := range conns {
		pos := strings.Index(xmlStr, cn)
		if pos < 0 {
			continue
		}
		x, y, w, h := getCoordsNear(xmlStr, pos+len(cn)/2)
		if x == 0 && y == 0 && w == 0 && h == 0 {
			continue
		}
		over := false
		for _, r := range rects {
			if overlap(x, y, w, h, r[0], r[1], r[2], r[3]) {
				over = true
				break
			}
		}
		if over {
			xmlStr = strings.Replace(xmlStr, cn, "", 1)
			removed++
		}
	}
	return xmlStr, removed
}

// main workflow: run extractor (writes meta.json), unzip, clean slides, rezip, optional libreoffice
func extractPPTXAndClean(pptxPath, outJSON, outCleanPptx string, makePDF bool) error {
	// 1) Run the extractor portion (this function extracts metadata + media files)
	// We'll reuse most of your extractPPTX logic, but return Result and files map so we can use coords.
	// For simplicity, call the same extraction code inline here to produce result and files map.

	// Open zip reader
	r, err := zip.OpenReader(pptxPath)
	if err != nil {
		return fmt.Errorf("cannot open pptx: %v", err)
	}
	defer r.Close()

	files := map[string]*zip.File{}
	for _, f := range r.File {
		files[f.Name] = f
	}

	var result Result
	result.Metadata = Metadata{}

	// read presentation size if available
	if presF, ok := files["ppt/presentation.xml"]; ok {
		presXML := readZipFile(presF)
		if m := sldSzRe.FindStringSubmatch(presXML); m != nil {
			if v, err := strconv.Atoi(m[1]); err == nil && v > 0 {
				slideWidthEMU = v
			}
			if v, err := strconv.Atoi(m[2]); err == nil && v > 0 {
				slideHeightEMU = v
			}
		}
	}

	// docProps
	if f, ok := files["docProps/core.xml"]; ok {
		data := readZipFile(f)
		if m := regexp.MustCompile(`<dc:creator>(.*?)</dc:creator>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Author = m[1]
		}
		if m := regexp.MustCompile(`<dc:title>(.*?)</dc:title>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Title = m[1]
		}
	}
	if f, ok := files["docProps/app.xml"]; ok {
		data := readZipFile(f)
		if m := regexp.MustCompile(`<Slides>(\d+)</Slides>`).FindStringSubmatch(data); m != nil {
			result.Metadata.SlideCount = m[1]
		}
		if m := regexp.MustCompile(`<Application>(.*?)</Application>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Application = m[1]
		}
		if m := regexp.MustCompile(`<Company>(.*?)</Company>`).FindStringSubmatch(data); m != nil {
			result.Metadata.Company = m[1]
		}
	}

	// ensure output_media dir
	os.MkdirAll("output_media", 0755)

	// parse slides
	for i := 1; ; i++ {
		slidePath := fmt.Sprintf("ppt/slides/slide%d.xml", i)
		f, ok := files[slidePath]
		if !ok {
			break
		}
		xmlStr := readZipFile(f)

		// relations for slide
		relsPath := fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels", i)
		rels := map[string]Relationship{}
		if rf, ok := files[relsPath]; ok {
			relsXML := readZipFile(rf)
			relRe := regexp.MustCompile(`<Relationship[^>]*Id="([^"]+)"[^>]*Type="([^"]+)"[^>]*Target="([^"]+)"[^>]*\/?>`)
			for _, m := range relRe.FindAllStringSubmatch(relsXML, -1) {
				rels[m[1]] = Relationship{Id: m[1], Type: m[2], Target: m[3]}
			}
		}

		var links []Link
		var mediaList []Media

		// extract text shapes as links (existing logic)
		for _, sp := range spRe.FindAllString(xmlStr, -1) {
			text := ""
			if m := regexp.MustCompile(`<a:t>(.*?)</a:t>`).FindAllStringSubmatch(sp, -1); m != nil {
				for _, mm := range m {
					text += mm[1]
				}
			}
			href := ""
			if m := regexp.MustCompile(`a:hlinkClick[^>]*r:id="(rId\d+)"`).FindStringSubmatch(sp); m != nil {
				if rel, ok := rels[m[1]]; ok {
					href = rel.Target
				}
			} else if m := regexp.MustCompile(`r:id="(rId\d+)"`).FindStringSubmatch(sp); m != nil {
				if rel, ok := rels[m[1]]; ok && strings.Contains(strings.ToLower(rel.Type), "hyperlink") {
					href = rel.Target
				}
			}
			x, y, w, h := getCoordsNear(xmlStr, strings.Index(xmlStr, sp)+len(sp)/2)
			if href != "" || text != "" {
				links = append(links, Link{Href: href, Text: text, X: x, Y: y, Width: w, Height: h})
			}
		}

		// extract media
		for _, match := range rIdRe.FindAllStringSubmatchIndex(xmlStr, -1) {
			if len(match) < 4 {
				continue
			}
			capStart := match[2]
			capEnd := match[3]
			rid := xmlStr[capStart:capEnd]
			matchPos := match[0]
			x, y, w, h := getCoordsNear(xmlStr, matchPos)

			if rel, ok := reMapLookup(files, rid); ok {
				ltype := strings.ToLower(rel.Type)
				ext := strings.ToLower(filepath.Ext(rel.Target))
				mediaKind := ""
				if strings.Contains(ltype, "image") || ext == ".png" || ext == ".jpg" || ext == ".jpeg" || ext == ".gif" || ext == ".bmp" {
					mediaKind = "image"
				} else if strings.Contains(ltype, "video") || ext == ".mp4" || ext == ".mov" || ext == ".avi" || ext == ".wmv" || ext == ".flv" {
					mediaKind = "video"
				} else if strings.Contains(ltype, "audio") || ext == ".mp3" || ext == ".wav" || ext == ".m4a" || ext == ".aac" {
					mediaKind = "audio"
				} else {
					if strings.Contains(strings.ToLower(rel.Target), "media/") {
						switch ext {
						case ".png", ".jpg", ".jpeg", ".gif", ".bmp":
							mediaKind = "image"
						case ".mp4", ".mov", ".avi", ".wmv", ".flv":
							mediaKind = "video"
						case ".mp3", ".wav", ".m4a", ".aac":
							mediaKind = "audio"
						default:
							// skip
						}
					}
				}

				if mediaKind != "" {
					local := exportFile(rel.Target, files, "output_media")
					if local != "" {
						mediaList = append(mediaList, Media{
							File:   local,
							Type:   mediaKind,
							X:      x,
							Y:      y,
							Width:  w,
							Height: h,
						})
					}
				}
			}
		}

		result.Slides = append(result.Slides, Slide{
			Slide: fmt.Sprint(i),
			Links: links,
			Media: mediaList,
		})
	}

	// write meta json
	if outJSON != "" {
		out, _ := json.MarshalIndent(result, "", "  ")
		_ = os.WriteFile(outJSON, out, 0644)
	}

	// 2) unzip pptx to temp dir for modification
	tmpDir, err := os.MkdirTemp("", "pptx_clean_*")
	if err != nil {
		return fmt.Errorf("tempdir: %v", err)
	}
	defer os.RemoveAll(tmpDir)

	if err := unzip(pptxPath, tmpDir); err != nil {
		return fmt.Errorf("unzip: %v", err)
	}

	// 3) for each slide, build rects from links+media and remove overlapping shapes/connectors
	totalRemoved := 0
	for _, s := range result.Slides {
		slideFile := filepath.Join(tmpDir, "ppt", "slides", fmt.Sprintf("slide%s.xml", s.Slide))
		b, err := os.ReadFile(slideFile)
		if err != nil {
			// skip if slide file not found on disk (shouldn't happen)
			continue
		}
		xmlStr := string(b)

		// build rect list
		var rects [][4]int
		for _, l := range s.Links {
			if l.Width > 0 && l.Height > 0 {
				rects = append(rects, [4]int{l.X, l.Y, l.Width, l.Height})
			}
		}
		for _, m := range s.Media {
			if m.Width > 0 && m.Height > 0 {
				rects = append(rects, [4]int{m.X, m.Y, m.Width, m.Height})
			}
		}
		if len(rects) == 0 {
			continue
		}

		// remove shapes that overlap rects
		updated, removed := removeShapesFromSlideXML(xmlStr, rects)
		if removed > 0 {
			totalRemoved += removed
			_ = os.WriteFile(slideFile, []byte(updated), 0644)
			fmt.Printf("Slide %s: removed %d shapes/connectors\n", s.Slide, removed)
		}
	}

	// 4) rezip tmpDir into outCleanPptx
	if outCleanPptx == "" {
		outCleanPptx = "cleaned_" + filepath.Base(pptxPath)
	}
	if err := zipDir(tmpDir, outCleanPptx); err != nil {
		return fmt.Errorf("zipDir: %v", err)
	}

	// 5) optionally convert to PDF
	if makePDF {
		if err := convertWithLibreOffice(outCleanPptx); err != nil {
			fmt.Println("LibreOffice convert failed:", err)
		}
	}

	fmt.Printf("Cleaning done. Removed total %d shapes. Cleaned PPTX: %s\n", totalRemoved, outCleanPptx)
	return nil
}

// helper to find rel by rId across rels files
func reMapLookup(files map[string]*zip.File, rid string) (Relationship, bool) {
	for path, f := range files {
		if !strings.HasSuffix(path, ".rels") {
			continue
		}
		data := readZipFile(f)
		relRe := regexp.MustCompile(`<Relationship[^>]*Id="(` + regexp.QuoteMeta(rid) + `)"[^>]*Type="([^"]+)"[^>]*Target="([^"]+)"[^>]*\/?>`)
		if m := relRe.FindStringSubmatch(data); m != nil {
			return Relationship{Id: m[1], Type: m[2], Target: m[3]}, true
		}
	}
	return Relationship{}, false
}

// convert with libreoffice
func convertWithLibreOffice(pptxPath string) error {
	dir := filepath.Dir(pptxPath)
	cmd := exec.Command("libreoffice", "--headless", "--convert-to", "pdf", pptxPath, "--outdir", dir)
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("libreoffice error: %v - %s", err, stderr.String())
	}
	return nil
}

func main() {
	if len(os.Args) < 4 {
		fmt.Println("Usage: go run pptx_clean_and_extract.go <input.pptx> <meta.json> <out_cleaned.pptx> [--pdf]")
		return
	}
	input := os.Args[1]
	meta := os.Args[2]
	outClean := os.Args[3]
	makePDF := false
	if len(os.Args) >= 5 && os.Args[4] == "--pdf" {
		makePDF = true
	}

	if err := extractPPTXAndClean(input, meta, outClean, makePDF); err != nil {
		fmt.Println("Error:", err)
		return
	}
}
