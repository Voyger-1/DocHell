// Slide XML structs
type Slide struct {
	XMLName xml.Name `xml:"sld"`
	Bg      *Bg      `xml:"cSld>bg"`
}

type Bg struct {
	BlipFill *BlipFill `xml:"blipFill"`
}

type BlipFill struct {
	Blip Blip `xml:"blip"`
}

type Blip struct {
	Embed string `xml:"embed,attr"` // r:embed
}

func main() {
	pptxFile := "example.pptx" // your PPTX file
	slideIndex := 1             // parse slide1.xml

	r, err := zip.OpenReader(pptxFile)
	if err != nil {
		log.Fatal(err)
	}
	defer r.Close()

	var slideXMLData []byte
	var relsXMLData []byte

	slidePath := fmt.Sprintf("ppt/slides/slide%d.xml", slideIndex)
	relsPath := fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels", slideIndex)

	// 1️⃣ Extract slide XML and rels XML from ZIP
	for _, f := range r.File {
		if f.Name == slidePath {
			rc, _ := f.Open()
			slideXMLData, _ = io.ReadAll(rc)
			rc.Close()
		}
		if f.Name == relsPath {
			rc, _ := f.Open()
			relsXMLData, _ = io.ReadAll(rc)
			rc.Close()
		}
	}

	if slideXMLData == nil {
		log.Fatal("Slide XML not found")
	}

	// 2️⃣ Parse slide XML
	var slide Slide
	if err := xml.Unmarshal(slideXMLData, &slide); err != nil {
		log.Fatal(err)
	}

	if slide.Bg != nil && slide.Bg.BlipFill != nil {
		embedID := slide.Bg.BlipFill.Blip.Embed
		fmt.Println("Background image r:embed ID:", embedID)

		// 3️⃣ Parse relationships to get actual media file
		type Relationship struct {
			ID     string `xml:"Id,attr"`
			Target string `xml:"Target,attr"`
		}
		type Relationships struct {
			XMLName xml.Name      `xml:"Relationships"`
			Rel     []Relationship `xml:"Relationship"`
		}

		var rels Relationships
		if err := xml.Unmarshal(relsXMLData, &rels); err != nil {
			log.Fatal(err)
		}

		for _, rel := range rels.Rel {
			if rel.ID == embedID {
				mediaPath := filepath.Join("ppt/slides", filepath.Dir("_rels/slide1.xml.rels"), rel.Target)
				mediaPath = strings.ReplaceAll(mediaPath, "\\", "/") // fix slashes
				fmt.Println("Background image path in PPTX ZIP:", mediaPath)
				break
			}
		}
	} else {
		fmt.Println("No background image found for this slide.")
	}
}

// // - Absolute layout faithful to slide size

// // - Audio/Video ➜ <audio>/<video> with extracted sources (mp3/mp4/webm/ogg), poster if available
